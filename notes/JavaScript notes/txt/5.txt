understanding functions - [parameters , arguments , rest parameters , hoisting , variable hoisting , function hoisting]
  //rest parameters
    ...a will collect all the remaining arguments into an array
    example:
      function test(a, b,...c) { //rest parameters
        console.log(a); //2
        console.log(c); //[4, 5, 6]
      }
      console.log(test(2, 3, 4, 5, 6));

  //parameters in JavaScript [required , default , destructured , rest]
  //destructured parameters
    example:
      function test({a, b, c}) { //destructured parameters
        console.log(a); //2
        console.log(b); //3
        console.log(c); //4
      }
      console.log(test({a: 2, b: 3, c: 4}));

  //arguments in JavaScript [positional, default, spread]
    //spread arguments
      example:
        function test(a, b, c) {
          console.log(a); //2
          console.log(b); //3
          console.log(c); //4
        }
        let arr = [2, 3, 4];
        console.log(test(...arr)); //spread operator

//more on functions in JavaScript
  arrow function - fat arrow function -> both are same
    ()=>{
      console.log("arrow function");
    }
  anonymous function - function without a name
    example:
      let anonFunc = function() { //anonymous function
        console.log("anonymous function");
      }
      anonFunc();

  higher order function -> 1. ek function jo ki return karde dursre function ko
                          2. ya fir ek function jo ki dusre function ko as a parameter le
    example:
      function higherOrderFunc(func) { //higher order function
        func();
      }
      function callbackFunc() { //callback function
        console.log("callback function executed");
      }
      higherOrderFunc(callbackFunc);
  callback function

  first class function -> jaha jaha variable use hota hai waha waha function bhi use ho sakta hai
    example:
      let firstClassFunc = function() { //first class function
        console.log("first class function");
      }
      firstClassFunc(); //function as a value

      -> function as parameter
      -> function as return value

  pure function
    same input = same output
    isme koi side effect nahi hota

  impure function
    same input = different output
    isme side effect ho sakta hai

//Closures in JavaScript -> ek function hai jo return karta hai function but returning function hai vo parrent funciton ka koi variable use kata hai
  example:
    function outerFunction() {
      let outerVariable = "I am from outer function";
      function innerFunction() {
        console.log(outerVariable); //accessing outer function variable
      }
      return innerFunction;
    }
    let closureFunction = outerFunction();
    closureFunction(); //I am from outer function
